# Integrated Plan: REST Migration + Twin/Monitor Consolidation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development

**Goal:** Migrate all 17 remaining tRPC routers to Hono REST routes with Zod v4 + @hono/zod-openapi. Consolidate twin + sim engines into a single module during migration. Replace web tRPC hooks with openapi-fetch.

**Architecture:** Hono REST replaces tRPC as the API layer. Zod v4 schemas with `.openapi()` metadata generate an OpenAPI spec, which drives `openapi-typescript` → `openapi-fetch` type-safe client for Web. Twin engine consolidation happens inline (build the unified REST routes against the merged engine). Zero dual-stack — REST replaces tRPC file by file.

**Tech Stack:** Hono v4 + @hono/zod-openapi + Zod v4 + openapi-fetch + openapi-typescript

```typescript
// apps/server/src/middleware/auth.ts
import { createMiddleware } from 'hono/factory'
import { verifyToken } from '../core/lib/jwt'
import type { JwtPayload } from '../core/lib/jwt'

export const jwtAuth = createMiddleware(async (c, next) => {
  const header = c.req.header('Authorization')
  if (!header?.startsWith('Bearer ')) {
    return c.json({ error: 'Unauthorized', message: 'Missing or invalid token' }, 401)
  }
  try {
    const payload = await verifyToken(header.slice(7))
    c.set('userId', payload.sub as string)
    c.set('userRole', payload.role as string)
    await next()
  } catch {
    return c.json({ error: 'Unauthorized', message: 'Invalid or expired token' }, 401)
  }
})
```

- [ ] **Step 2: Create Hono RBAC middleware**

```typescript
// apps/server/src/middleware/rbac.ts
import { createMiddleware } from 'hono/factory'
import { db } from '../core/db'
import { eq, and } from 'drizzle-orm'
import { permissions, rolePermissions } from '../core/db/schema/auth-ext'

const rolePermCache = new Map<string, Set<string>>()

async function loadPermissions(role: string): Promise<Set<string>> {
  if (rolePermCache.has(role)) return rolePermCache.get(role)!
  const rows = await db
    .select({ code: permissions.code })
    .from(rolePermissions)
    .innerJoin(permissions, eq(rolePermissions.permissionId, permissions.id))
    .where(eq(rolePermissions.role, role))
  const set = new Set(rows.map((r) => r.code))
  rolePermCache.set(role, set)
  return set
}

export function requirePermission(...codes: string[]) {
  return createMiddleware(async (c, next) => {
    const role = c.get('userRole')
    if (!role) return c.json({ error: 'Forbidden' }, 403)
    if (role === 'super_admin') return await next()
    const perms = await loadPermissions(role)
    if (codes.every((code) => perms.has(code))) return await next()
    return c.json({ error: 'Forbidden', message: 'Insufficient permissions' }, 403)
  })
}
```

// apps/server/src/routes/auth.ts
import { OpenAPIHono, createRoute, z } from '@hono/zod-openapi'
import { db } from '../core/db'
import { users } from '../core/db/schema'
import { eq } from 'drizzle-orm'
import { hashPassword, verifyPassword } from '../core/lib/password'
import { signToken, signRefreshToken } from '../core/lib/jwt'

const auth = new OpenAPIHono()

const registerRoute = createRoute({
  method: 'post',
  path: '/register',
  request: {
    body: { content: { 'application/json': { schema: z.object({
      username: z.string().min(3).max(50).openapi({ example: 'researcher' }),
      password: z.string().min(6).openapi({ example: 'secret123' }),
      displayName: z.string().optional().openapi({ example: '研究员' }),
    }) } } },
  },
  responses: { 201: { description: 'Created' }, 409: { description: 'Conflict' } },
})

auth.openapi(registerRoute, async (c) => {
  const body = c.req.valid('json')
  const existing = await db.select().from(users).where(eq(users.username, body.username)).limit(1)
  if (existing.length) return c.json({ error: 'Username taken' }, 409)
  const [user] = await db.insert(users).values({
    username: body.username,
    passwordHash: await hashPassword(body.password),
    displayName: body.displayName ?? body.username,
    role: 'user',
  }).returning()
  return c.json({ id: user.id, username: user.username }, 201)
})

const loginRoute = createRoute({
  method: 'post',
  path: '/login',
  request: {
    body: { content: { 'application/json': { schema: z.object({
      username: z.string().openapi({ example: 'admin' }),
      password: z.string().openapi({ example: 'admin123' }),
    }) } } },
  },
  responses: { 200: { description: 'OK' }, 401: { description: 'Unauthorized' } },
})

auth.openapi(loginRoute, async (c) => {
  const body = c.req.valid('json')
  const [user] = await db.select().from(users).where(eq(users.username, body.username)).limit(1)
  if (!user) return c.json({ error: 'Invalid credentials' }, 401)
  const valid = await verifyPassword(user.passwordHash, body.password)
  if (!valid) return c.json({ error: 'Invalid credentials' }, 401)
  const accessToken = await signToken({ sub: user.id, role: user.role })
  const refreshToken = await signRefreshToken({ sub: user.id })
  return c.json({ accessToken, refreshToken, user: { id: user.id, username: user.username, role: user.role, displayName: user.displayName } })
})

const refreshRoute = createRoute({
  method: 'post',
  path: '/refresh',
  request: {
    body: { content: { 'application/json': { schema: z.object({
      refreshToken: z.string().openapi({ example: '...' }),
    }) } } },
  },
  responses: { 200: { description: 'OK' }, 401: { description: 'Unauthorized' } },
})

auth.openapi(refreshRoute, async (c) => {
  const body = c.req.valid('json')
  // Standard refresh token rotation logic — verify old token, issue new pair
  return c.json({ message: 'Not yet implemented' }, 501)
})

export { auth }

  const safe = rows.map(({ passwordHash, ...rest }) => rest)
  return c.json(safe)
})

const meRoute = createRoute({
  method: 'get',
  path: '/me',
  responses: { 200: { description: 'Current user' } },
})

usersApp.openapi(meRoute, async (c) => {
  const userId = c.get('userId')
  const [user] = await db.select().from(users).where(eq(users.id, userId)).limit(1)
  if (!user) return c.json({ error: 'Not found' }, 404)
  const { passwordHash, ...safe } = user
  return c.json(safe)
})

const updateRoute = createRoute({
  method: 'patch',
  path: '/:id',
  request: {
    body: { content: { 'application/json': { schema: z.object({
      displayName: z.string().optional(),
      phone: z.string().optional(),
      email: z.string().optional(),
    }) } } },
  },
  responses: { 200: { description: 'Updated' } },
})

usersApp.openapi(updateRoute, async (c) => {
  const id = c.req.param('id')
  const body = c.req.valid('json')
  const [updated] = await db.update(users).set(body).where(eq(users.id, id)).returning()
  if (!updated) return c.json({ error: 'Not found' }, 404)
  const { passwordHash, ...safe } = updated
  return c.json(safe)
})

export { usersApp }

### Strategy: Page-by-page migration

Each page that currently does `trpc.xxx.useQuery(...)` gets rewritten to use `api.GET(...)` or `api.POST(...)`.

### Step 1: Generate client types

```bash
cd apps/web
pnpm add openapi-fetch openapi-typescript
# Start server, then:
npx openapi-typescript http://localhost:3000/openapi.json -o src/api/types.ts
# Add src/api/types.ts to .gitignore
```

### Step 2: Create API client

```typescript
// apps/web/src/api/client.ts
import createClient from 'openapi-fetch'
import type { paths } from './types'

const baseUrl = import.meta.env.VITE_API_BASE || 'http://localhost:3000'

export const api = createClient<paths>({
  baseUrl,
  headers: () => {
    const token = localStorage.getItem('token')
    return token ? { Authorization: `Bearer ${token}` } : {}
  },
})
```
